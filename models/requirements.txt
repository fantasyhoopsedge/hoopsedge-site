# Python deps for the projection models (models/*). This layer runs OUTSIDE Next
# and outside npm — it is the only Python in the repo.
#
#   python -m venv .venv
#   .venv/Scripts/activate        # Windows;  source .venv/bin/activate on macOS/Linux
#   pip install -r models/requirements.txt
#
# Developed and validated against Python 3.14 with the versions below. Pins are
# floored at what the models were actually run on and capped at the next major,
# since a silent numpy/pandas major bump can change dtype and NaN handling
# underneath the feature builders.
#
# pyarrow is not imported directly anywhere — it is the engine pandas.read_parquet
# needs for the hoopR player-box feed, and omitting it fails only at runtime.

numpy>=2.5,<3
pandas>=3.0,<4
scikit-learn>=1.9,<2
# scipy is a scikit-learn dependency already, but Stage 6 (models/validation) imports
# scipy.stats.spearmanr directly, so it is declared here rather than relied on transitively.
scipy>=1.13,<2
pyarrow>=25.0,<26
